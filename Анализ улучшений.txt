2. Prisma модель:

model widget_configs {
  id            String   @id @default(cuid())
  website_token String   @unique
  name          String
  welcome_text  String?
  primary_color String   @default("#3b82f6")
  position      String   @default("right") // 'right' | 'left'
  enabled       Boolean  @default(true)
  allowed_domains String[] // CORS
  created_at    DateTime @default(now())
  updated_at    DateTime @updatedAt
}

model widget_conversations {
  id            String   @id @default(cuid())
  widget_config_id String
  widget_config widget_configs @relation(fields: [widget_config_id], references: [id])
  visitor_id    String   // fingerprint/cookie
  visitor_name  String?
  visitor_email String?
  status        String   @default("open") // 'open' | 'closed' | 'pending'
  assigned_to   Int?
  created_at    DateTime @default(now())
  updated_at    DateTime @updatedAt

  @@index([widget_config_id])
  @@index([visitor_id])
}
