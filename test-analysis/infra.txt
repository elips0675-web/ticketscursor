===== FILE: scripts/test-e2e-api.js =====
// E2E API тесты через Node.js fetch
// Проверяет основные эндпоинты без зависимостей от Playwright

const BASE_URL = 'http://localhost:4000'

async function test(name, fn) {
  try {
    await fn()
    console.log(`✓ ${name}`)
  } catch (e) {
    console.error(`✗ ${name}: ${e.message}`)
    process.exitCode = 1
  }
}

function assert(condition, msg) {
  if (!condition) throw new Error(msg)
}

async function api(method, path, body, headers = {}) {
  const res = await fetch(`${BASE_URL}${path}`, {
    method,
    headers: { 'Content-Type': 'application/json', ...headers },
    body: body ? JSON.stringify(body) : undefined,
  })
  return { status: res.status, data: await res.json() }
}

// Health check
await test('GET /api/health', async () => {
  const { status, data } = await api('GET', '/api/health')
  assert(status === 200, `Expected 200, got ${status}`)
  assert(data.success === true, 'Expected success: true')
})

// Auth
let token
await test('POST /api/auth/dev-login', async () => {
  const { status, data } = await api('POST', '/api/auth/dev-login', null, {
    'X-Email': 'alexey@example.com',
  })
  assert(status === 200, `Expected 200, got ${status}`)
  token = data.data.token || data.token
  assert(token, 'Expected token')
})

await test('GET /api/auth/me', async () => {
  const { status } = await api('GET', '/api/auth/me', null, {
    Authorization: `Bearer ${token}`,
  })
  assert(status === 200, `Expected 200, got ${status}`)
})

// Tickets
await test('GET /api/tickets', async () => {
  const { status } = await api('GET', '/api/tickets', null, {
    Authorization: `Bearer ${token}`,
  })
  assert(status === 200, `Expected 200, got ${status}`)
})

// RBAC
await test('GET /api/admin/tags (403 for non-admin)', async () => {
  const { status } = await api('GET', '/api/admin/tags', null, {
    Authorization: `Bearer ${token}`,
  })
  assert(status === 403, `Expected 403, got ${status}`)
})

// i18n
await test('GET /api/auth/me with Accept-Language: ru', async () => {
  const { status } = await api('GET', '/api/auth/me', null, {
    Authorization: `Bearer ${token}`,
    'Accept-Language': 'ru',
  })
  assert(status === 200, `Expected 200, got ${status}`)
})

console.log('\n✓ All E2E API tests passed')

===== FILE: scripts/check-bundle-size.js =====
/* global console, process */
// Проверяет размер бандла Vite (max 2MB gzipped)

import { readFileSync, readdirSync } from 'fs'
import { join } from 'path'

const distDir = join(process.cwd(), 'dist')
const maxSizeKB = 2048 // 2MB

try {
  const files = readdirSync(join(distDir, 'assets')).filter(f => f.endsWith('.js'))
  let totalSize = 0

  for (const file of files) {
    const filePath = join(distDir, 'assets', file)
    const content = readFileSync(filePath)
    const gzipSize = require('zlib').gzipSync(content).length
    totalSize += gzipSize
    console.log(`${file}: ${(gzipSize / 1024).toFixed(1)}KB (gzipped)`)
  }

  console.log(`\nTotal: ${(totalSize / 1024).toFixed(1)}KB / ${maxSizeKB}KB limit`)

  if (totalSize > maxSizeKB * 1024) {
    console.error('❌ Bundle size exceeds limit!')
    process.exit(1)
  } else {
    console.log('✅ Bundle size OK')
  }
} catch (e) {
  console.warn('⚠️  Could not check bundle size:', e.message)
}

// Заполнение базы данных тестовыми данными для разработки

import { PrismaClient } from '@prisma/client'

const prisma = new PrismaClient()

async function main() {
  console.log('🌱 Seeding database...')

  // Создаём тестового пользователя
  const user = await prisma.user.upsert({
    where: { email: 'alexey@example.com' },
    update: {},
    create: {
      email: 'alexey@example.com',
      name: 'Алексей Тестов',
      role: 'admin',
    },
  })

  console.log('✅ User created:', user.email)

  // Создаём тестовые тикеты
  for (let i = 1; i <= 5; i++) {
    await prisma.ticket.create({
      data: {
        title: `Тестовый тикет #${i}`,
        description: `Описание тестового тикета ${i}`,
        priority: 'medium',
        status: 'open',
        creatorId: user.id,
      },
    })
  }

  console.log('✅ 5 test tickets created')
}

main()
  .catch((e) => {
    console.error(e)
    process.exit(1)
  })
  .finally(async () => {
    await prisma.$disconnect()
  })

===== FILE: scripts/bump-version.js =====
// Автоматическое обновление версии в package.json

import { readFileSync, writeFileSync } from 'fs'
import { join } from 'path'

const rootPkg = join(process.cwd(), 'package.json')
const serverPkg = join(process.cwd(), 'server', 'package.json')

const root = JSON.parse(readFileSync(rootPkg, 'utf-8'))
const server = JSON.parse(readFileSync(serverPkg, 'utf-8'))

// Обновляем патч-версию
const [major, minor, patch] = root.version.split('.').map(Number)
const newVersion = `${major}.${minor}.${patch + 1}`

root.version = newVersion
server.version = newVersion

writeFileSync(rootPkg, JSON.stringify(root, null, 2) + '\n')
writeFileSync(serverPkg, JSON.stringify(server, null, 2) + '\n')

console.log(`✅ Version bumped to ${newVersion}`)

// Тестирование аналитики (проверка метрик)

import { PrismaClient } from '@prisma/client'

const prisma = new PrismaClient()

async function main() {
  console.log('📊 Testing analytics...')

  // Проверяем количество тикетов по статусам
  const ticketsByStatus = await prisma.ticket.groupBy({
    by: ['status'],
    _count: true,
  })

  console.log('Tickets by status:')
  for (const group of ticketsByStatus) {
    console.log(`  ${group.status}: ${group._count}`)
  }

  // Проверяем количество тикетов по приоритетам
  const ticketsByPriority = await prisma.ticket.groupBy({
    by: ['priority'],
    _count: true,
  })

  console.log('Tickets by priority:')
  for (const group of ticketsByPriority) {
    console.log(`  ${group.priority}: ${group._count}`)
  }

  // Проверяем среднее время ответа
  const avgResponseTime = await prisma.ticket.aggregate({
    _avg: {
      responseTime: true,
    },
  })

  console.log('Average response time:', avgResponseTime._avg.responseTime)
}

main()
  .catch((e) => {
    console.error(e)
    process.exit(1)
  })
  .finally(async () => {
    await prisma.$disconnect()
  })
