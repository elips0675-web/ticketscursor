===== FILE: test/load/tickets.k6.js =====
import http from 'k6/http'
import { check, sleep } from 'k6'

const BASE_URL = __ENV.BASE_URL || 'http://localhost:4000'

export const options = {
  stages: [
    { duration: '30s', target: 10 },
    { duration: '1m', target: 20 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    http_req_duration: ['p(95)<500'],
    http_req_failed: ['rate<0.1'],
  },
}

export default function () {
  const loginRes = http.post(`${BASE_URL}/api/auth/dev-login`, null, {
    headers: { 'X-Email': 'alexey@example.com' },
  })

  check(loginRes, {
    'login successful': (r) => r.status === 200,
  })

  const token = loginRes.json('data.token') || loginRes.json('token')

  const ticketsRes = http.get(`${BASE_URL}/api/tickets`, {
    headers: { Authorization: `Bearer ${token}` },
  })

  check(ticketsRes, {
    'tickets status 200': (r) => r.status === 200,
  })

  sleep(1)
}

===== FILE: test/load/chat.k6.js =====
import http from 'k6/http'
import { check, sleep } from 'k6'

const BASE_URL = __ENV.BASE_URL || 'http://localhost:4000'
const TICKET_ID = __ENV.TICKET_ID || 'test-ticket-id'

export const options = {
  stages: [
    { duration: '30s', target: 10 },
    { duration: '1m', target: 20 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    http_req_duration: ['p(95)<500'],
    http_req_failed: ['rate<0.1'],
  },
}

export default function () {
  const loginRes = http.post(`${BASE_URL}/api/auth/dev-login`, null, {
    headers: { 'X-Email': 'alexey@example.com' },
  })

  check(loginRes, {
    'login successful': (r) => r.status === 200,
  })

  const token = loginRes.json('data.token') || loginRes.json('token')

  const commentsRes = http.get(
    `${BASE_URL}/api/tickets/${TICKET_ID}/comments?limit=20`,
    {
      headers: { Authorization: `Bearer ${token}` },
    }
  )

  check(commentsRes, {
    'comments status 200': (r) => r.status === 200,
  })

  sleep(1)
}

===== FILE: test/load/artillery.yml =====
config:
  target: 'http://localhost:4000'
  phases:
    - duration: 60
      arrivalRate: 10
  defaults:
    headers:
      Content-Type: 'application/json'

scenarios:
  - name: 'Create and read tickets'
    flow:
      - post:
          url: '/api/auth/dev-login'
          headers:
            X-Email: 'alexey@example.com'
          capture:
            - json: '$.data.token'
              as: 'token'
      - get:
          url: '/api/tickets'
          headers:
            Authorization: 'Bearer {{ token }}'
