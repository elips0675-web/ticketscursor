  # 1. Получить админский токен (dev): POST http://localhost:4000/api/auth/dev-login (в prod — 404, токен от администратора)

  k6 run -e TOKEN=$TOKEN test/load/tickets.k6.js

  k6 run -e TOKEN=$TOKEN -e CHAT_ID=1 test/load/chat.k6.js
  # 4. Тест wiki (100 concurrent, список + поиск)
  k6 run -e TOKEN=$TOKEN test/load/wiki.k6.js

  k6 run -e TOKEN=$TOKEN test/load/search.k6.js
  # 6. Тест новостей (100 concurrent)
  k6 run -e TOKEN=$TOKEN test/load/news.k6.js

===== FILE: test/load/tickets.k6.js =====
import http from 'k6/http'
import { check, sleep } from 'k6'
import { Rate, Trend } from 'k6/metrics'

const baseUrl = __ENV.API_URL || 'http://localhost:4000'
const token = __ENV.TOKEN

const ticketTrend = new Trend('ticket_list_duration')
const ticketErrors = new Rate('ticket_errors')

export const options = {
  stages: [
    { duration: '30s', target: 20 },
    { duration: '1m', target: 100 },
    { duration: '30s', target: 200 },
    { duration: '1m', target: 200 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    ticket_errors: ['rate<0.01'],
    http_req_duration: ['p(95)<1000'],
  },
}

const headers = {
  headers: { Authorization: `Bearer ${token}` },
}

export default function () {
  const listRes = http.get(`${baseUrl}/api/tickets?page=1&limit=20`, headers)
  ticketTrend.add(listRes.timings.duration)
  ticketErrors.add(listRes.status !== 200)

  check(listRes, {
    'tickets list status 200': (r) => r.status === 200,
    'tickets has data': (r) => Array.isArray(r.json('data')),
  })

  const payload = JSON.stringify({
    title: `Load test ticket ${Date.now()}`,
    description: 'Generated by k6 load test',
    priority: 'low',
    category: 'support',
  })
  const createRes = http.post(`${baseUrl}/api/tickets`, payload, {
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${token}`,
      'Idempotency-Key': `${__VU}-${Date.now()}`,
    },
  })
  ticketErrors.add(createRes.status !== 201)

  check(createRes, {
    'ticket created status 201': (r) => r.status === 201,
  })

  sleep(1)
}

===== FILE: test/load/chat.k6.js =====
import http from 'k6/http'
import { check, sleep } from 'k6'
import { Rate, Trend } from 'k6/metrics'

const baseUrl = __ENV.API_URL || 'http://localhost:4000'
const token = __ENV.TOKEN

const msgDuration = new Trend('chat_msg_duration')
const msgErrors = new Rate('chat_msg_errors')

export const options = {
  stages: [
    { duration: '30s', target: 10 },
    { duration: '1m', target: 50 },
    { duration: '30s', target: 100 },
    { duration: '1m', target: 100 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    chat_msg_errors: ['rate<0.05'],
    http_req_duration: ['p(95)<500'],
  },
}

const chatId = __ENV.CHAT_ID || 1

export default function () {
  const payload = JSON.stringify({ text: `Load test message ${Date.now()}` })
  const params = {
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${token}`,
      'Idempotency-Key': `${__VU}-${Date.now()}`,
    },
  }

  const res = http.post(`${baseUrl}/api/chats/${chatId}/messages`, payload, params)
  msgDuration.add(res.timings.duration)
  msgErrors.add(res.status !== 201)

  check(res, {
    'status is 201': (r) => r.status === 201,
    'has success': (r) => r.json('success') === true,
  })

  sleep(0.5)
}

import http from 'k6/http'
import { check, sleep } from 'k6'
import { Rate, Trend } from 'k6/metrics'

const baseUrl = __ENV.API_URL || 'http://localhost:4000'
const token = __ENV.TOKEN

const searchTrend = new Trend('search_duration')
const searchErrors = new Rate('search_errors')

export const options = {
  stages: [
    { duration: '30s', target: 10 },
    { duration: '1m', target: 50 },
    { duration: '30s', target: 100 },
    { duration: '1m', target: 100 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    search_errors: ['rate<0.02'],
    http_req_duration: ['p(95)<1000'],
  },
}

const headers = {
  headers: { Authorization: `Bearer ${token}` },
}

const queries = ['тикет', 'тест', 'ошибка', 'запрос', 'помощь']

export default function () {
  const q = queries[Math.floor(Math.random() * queries.length)]
  const res = http.get(`${baseUrl}/api/search?q=${encodeURIComponent(q)}`, headers)
  searchTrend.add(res.timings.duration)
  searchErrors.add(res.status !== 200)

  check(res, {
    'search status 200': (r) => r.status === 200,
    'search has results': (r) => typeof r.json('data') === 'object',
  })

  sleep(0.5)
}
